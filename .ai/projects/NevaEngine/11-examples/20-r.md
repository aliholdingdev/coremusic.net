## §1 Bağlam
R, Neva Engine'de **istatistik/ölçüm** dilidir (offline analiz · raporlama). Örnekler yalnız base R; harici paket adları `⚠️` ile işaretlidir, API'leri garanti edilmez. Tek iş parçacığı (R10) — DSP/I/O yolu K1-K6 (32-bit float · 44.1–192 kHz · varsayılan 48 kHz) üzerinden akar, canlı taraf C++ çekirdektedir.
## §2 Örnek 1 — Biquad EQ · (R16 · K2)
```r
# neva_biquad.R — R numeric = double (K2 sapması: float32 saha 64-bit genişler)
make_peaking <- function(f0, q, gain_db, fs) {
  q <- min(max(q, 0.05), 32.0); f0 <- min(max(f0, 10.0), 0.49 * fs)   # Nyquist
  a <- 10^(gain_db / 40); w0 <- 2 * pi * (f0 / fs); al <- sin(w0) / (2 * q); a0 <- 1 + al / a
  list(b = c(1 + al * a, -2 * cos(w0), 1 - al * a) / a0, a = c(1, -2 * cos(w0) / a0, (1 - al / a) / a0))
}
biquad_df1 <- function(coef, x) {                 # DF1 4 durum; durum çağrıda sıfır
  y <- numeric(length(x)); x1 <- x2 <- y1 <- y2 <- 0
  for (n in seq_along(x)) {
    y[n] <- coef$b[1]*x[n] + coef$b[2]*x1 + coef$b[3]*x2 - coef$a[2]*y1 - coef$a[3]*y2; x2 <- x1; x1 <- x[n]; y2 <- y1; y1 <- y[n]
  }
  y
}
```
**Neva bağlamı:** offline katsayı doğrulama — koefisyonlar C++20 çekirdeğe (doc 02/03) aktarılır; üretim DF2T R'de değil.
## §3 Örnek 2 — 32f buffer döngüsü · (K2 · K4 · K5)
```r
K4_RATES <- c(44100, 48000, 88200, 96000, 176400, 192000)
pick_sample_rate <- function(preferred, device_rates) {             # K4 → K6 → K5
  pref <- if (preferred %in% K4_RATES) preferred else 48000         # K4 default
  if (pref %in% device_rates) return(pref)                          # K6 SR eşit
  k4 <- device_rates[device_rates %in% K4_RATES]; if (length(k4) == 0) 48000 else max(k4)   # K5
}
pad_and_clip <- function(x, frame, ceiling_db = -0.1) {  # vektörel alternatif
  rem <- length(x) %% frame                                        # K5 dolgu
  padded <- if (rem == 0) x else c(x, numeric(frame - rem)); c1 <- 10^(ceiling_db / 20)
  pmax(-c1, pmin(c1, padded))                                      # clip (K2)
}
```
**Sapma (madde-başına):** vektörel alternatif `pmin/pmax` (clip) + `c()` (pad); IIR durum-geçişi sıralı → EQ vektörleştiremez. **Neva bağlamı:** offline blok işleme — canlı callback değil.
## §4 Örnek 3 — SPSC ring buffer · (R10)
```r
# SAPMA: R tek iş parçacığı — gerçek lock-free SPSC YOK (R10); RBEB emülasyonu
simple_ring <- function(cap) {                                     # cap = 2^n
  e <- new.env(parent = emptyenv())
  e$buf <- numeric(cap); e$mask <- cap - 1L; e$head <- 0L; e$tail <- 0L; e
}
ring_push <- function(r, v) {                             # FALSE = dolu
  nxt <- (r$head + 1L) %% (r$mask + 1L); if (nxt == r$tail) return(FALSE)
  r$buf[r$head + 1L] <- v; r$head <- nxt; TRUE            # 1-bazlı indeks (+1)
}
ring_pop <- function(r) {                                 # NULL = boş
  if (r$tail == r$head) return(NULL); v <- r$buf[r$tail + 1L]; r$tail <- (r$tail + 1L) %% (r$mask + 1L); v
}
```
**Dürüst sapma:** lock-free/atomik SPSC R'de YOK (tek thread). **Neva bağlamı:** çok-thread gerekiyorsa `parallel`/`later` paketleri (`⚠️ VERIFICATION REQUIRED`, RT değil); gerçek SPSC C++/Rust çekirdektedir (R10 · doc 03).
## §5 Örnek 4 — WAV 32f I/O · (K3 · K1)
```r
# base R'de WAV okuyucu YOK → elle 44-bayt sabit-ofset parse (K3 · 32f mono) · harici paket örn. tuneR/seewave (⚠️ VERIFICATION REQUIRED)
read_wav_f32 <- function(path) {
  con <- file(path, "rb"); on.exit(close(con))
  if (rawToChar(readBin(con, "raw", 4)) != "RIFF") stop("RIFF degil")
  seek(con, 8); if (rawToChar(readBin(con, "raw", 4)) != "WAVE") stop("WAVE degil")
  seek(con, 20); fmt <- readBin(con, "integer", 1, size = 2, signed = FALSE)
  ch <- readBin(con, "integer", 1, size = 2, signed = FALSE); fs <- readBin(con, "integer", 1, size = 4, signed = FALSE)
  seek(con, 34); bits <- readBin(con, "integer", 1, size = 2, signed = FALSE)
  if (fmt != 3L || ch != 1L || bits != 32L) stop("yalniz 32f mono (K1·K3)")
  seek(con, 40); nd <- readBin(con, "integer", 1, size = 4, signed = FALSE); seek(con, 44)
  list(pcm = readBin(con, "double", nd / 4L, size = 4), fs = fs, ch = ch)  # size=4 → f32→double
}
```
**Dürüst sapma:** elle sabit-ofset parse — genişletilmiş chunk'larda chunk-walk gerekir; `readBin` `size = 4` host byte sırasına bağlı (x86 LE). **Neva bağlamı:** offline test I/O — doc 05 PCM doğrulaması (K1 girdi · K3 çıktı).
## §6 Örnek 5 — Compressor+limiter · (R10 · K3)
```r
# envelope (atk 5 ms / rel 50 ms) → −12 dB thr → 4:1 → clamp 0 dBFS (K3)
compressor <- function(fs) {
  e <- new.env(parent = emptyenv())
  e$env <- 0; e$ka <- exp(-1/(0.005*fs)); e$kr <- exp(-1/(0.050*fs))
  e$thr <- 10^(-12/20); e$ratio <- 4; e
}
comp_process <- function(c, x) {                  # tek thread (R10) — canlı chain C++'ta
  a <- if (abs(x) > c$env) c$ka else c$kr; c$env <- a*c$env + (1 - a)*abs(x)
  g <- if (c$env > c$thr) 10^(((20*log10(c$env + 1e-12) + 12) * (1 - 1/c$ratio))/20) else 1; min(max(x * g, -1), 1)   # brickwall
}
```
**Neva bağlamı:** offline dinamik doğrulama — üretim compressor C++ çekirdekte (R10); `env` durumu `new.env` ile kalıcı (K7).
## §7 Örnek 6 — OS/cihaz enum · (R18)
```r
# SAPMA: base R'de ses cihazı API'si YOK — os_str YOK (Sys.info() Windows-odaklı);
# backend enum metin dosyasından okunur (R18; doc 04 §4)
os_kind <- function(path = "/tmp/os.txt") {
  s <- if (file.exists(path)) tolower(readLines(path, n = 1, warn = FALSE)) else tolower(Sys.info()[["sysname"]])
  if (grepl("win", s)) "Windows" else if (grepl("darwin|mac", s)) "macOS" else if (grepl("linux", s)) "Linux" else "Unknown"
}
backend_from_name <- function(name) {                     # kayıt edilmiş seçim (R18)
  n <- tolower(name)
  if (grepl("wasapi", n)) "wasapi" else if (grepl("asio", n)) "asio" else if (grepl("coreaudio", n)) "coreaudio" else if (grepl("alsa", n)) "alsa"
  else if (grepl("pulse", n)) "pulse" else if (grepl("null", n)) "null_device" else "unknown"
}
```
**Dürüst sapma:** gerçek cihaz listesi/enum R'de YOK — profil/ayar katmanı; cihaz enum yalnız C++/Rust çekirdekte (R18 · doc 03 §7).
